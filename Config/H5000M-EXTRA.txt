# H5000M 机型专属插件层（Bandix Plus / OpenAppFilter）
#
# 为什么单独成层，而不是写进 Config/MT5700.txt 或各机型配置里：
#   - Config/MT5700.txt 是 MT 方案共用层，H5000M / AP3000M / X86 的 MT5700 变体
#     以及 NetWiz-MT5700 都会叠加它，写在那里等于把 H5000M 专属插件塞进其它机型；
#   - 逐份复制到 H5000M-WIFI-YES / -FM350 / -NETWIZ-* / -NETMONITOR 六个文件里，
#     改一处要同步六处，迟早漏。
# 因此由 WRT-CORE.yml 在 WRT_CONFIG 以 H5000M 开头时追加本文件（叠加在
# Config/<WRT_CONFIG>.txt + Config/GENERAL.txt 之后、Settings.sh / ApplyMTMode.sh
# 之前），非 H5000M 构建完全不加载。
#
# 本层只放「纯增量」的 =y 符号；不要在这里禁用任何包——禁用项应写进对应机型的
# 配置层，否则会影响同机型的其它变体。

# ===== Bandix Plus 多网口流量监控（timsaya/luci-app-bandix-plus）=====
#
# 后端 bandix-plus 是预编译二进制（aarch64-unknown-linux-musl，从上游 Release
# 下载，hash 按 ARCH 分支写在 Makefile 里），运行期依赖内核 BPF；前端 LUCI_DEPENDS
# 已含后端，这里显式选中后端便于在 menuconfig 里单独关闭。
# 取包逻辑见 Scripts/Packages.sh 的 H5000M 分支（前端仓库根目录与包目录同名，
# 不能走 UPDATE_PACKAGE 的 pkg/name/all 三种模式，需手工提取）。
CONFIG_PACKAGE_bandix-plus=y
CONFIG_PACKAGE_luci-app-bandix-plus=y
CONFIG_PACKAGE_luci-i18n-bandix-plus-zh-cn=y

# ===== OpenAppFilter 家长控制 / 应用过滤（destan19/OpenAppFilter）=====
#
# 三件套缺一不可：kmod-oaf 是 netfilter 模块（没有它 oafd 无法下发规则），
# appfilter 是用户态守护进程 oafd + feature.bin 特征库，luci-app-oaf 是界面。
# 前端为 Lua/CBI 实现，依赖 luci-compat + luci-lua-runtime（25.12 仍提供，
# 位于 luci feed 的 modules/ 下）。
#
# ⚠️ 与 feed 的冲突：immortalwrt/packages 里已有 net/open-app-filter 旧快照，
# 其单文件 Makefile 同样定义了 appfilter 与 kmod-oaf 两个包名。两套同名包共存
# 会让 OpenWrt 报 Duplicate package name，并在 make defconfig 阶段把整组符号连
# 依赖一起丢掉 —— 表现为「构建全绿、固件里既没有界面也没有 oafd」。
# 清 feed 的动作在 Scripts/Packages.sh 的 INSTALL_OPENAPPFILTER 里完成。
CONFIG_PACKAGE_kmod-oaf=y
CONFIG_PACKAGE_appfilter=y
CONFIG_PACKAGE_luci-app-oaf=y
CONFIG_PACKAGE_luci-i18n-oaf-zh-cn=y
