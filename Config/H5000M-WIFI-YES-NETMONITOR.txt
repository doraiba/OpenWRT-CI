# Hiveton H5000M 自用配置（NetMonitor 网络监控 + Taygedo 签到 + MT5700 方案）
#
# ⚠️ 本配置为【个人自用】配置：
#   - 仅服务作者自有的 H5000M 设备；
#   - 产物在 Releases 中单独归组「自用」（Release Tag 形如 自用-<YY.MM.DD>，
#     固件名后缀 -自用），不进入 MT5700 / MT5700M / FM350 / NetWiz / BASE
#     等其他任何分类。
#
# 在现有 MT5700 方案基础上新增两个 LianXia233 插件：
#   - luci-app-netmonitor —— 网络延迟 / 连通性实时监控
#     （https://github.com/LianXia233/luci-app-netmonitor）
#   - luci-app-taygedo    —— 塔吉多（幻塔 / 异环）每日自动签到
#     （https://github.com/LianXia233/taygedo-CI）
#
# MT 插件层刻意不写在本文件中：由调用方传入 MT_MODE=MT5700，由
# Scripts/ApplyMTMode.sh 在 Settings.sh 之后叠加 Config/MT5700.txt 并执行
# 互斥保护（MT5700M 侧包会被强制禁用）。
#
# 定时工作流（H5000M-NETMONITOR-AUTO）已绑定 MT_MODE=MT5700；手动触发
# WRT-BUILD 时必须把 MT Mode 下拉同步选为 MT5700，若留空 ApplyMTMode
# 空模式会禁用全部 MT 包，「校验 luci-app-netmonitor / luci-app-taygedo
# 已选中」步骤虽不校验 MT 包，但产物将缺少 MT5700 模组功能，名不副实。
#
# 命名保留 "H5000M-WIFI-YES" 子串以继承按机型名匹配的既有逻辑
#（WRT-CORE 中 AP3000M 专属步骤靠 contains 命中，本配置为 H5000M 不受影响）。
# 设备平台
CONFIG_TARGET_mediatek=y
CONFIG_TARGET_mediatek_filogic=y

# 仅编译 Hiveton H5000M
CONFIG_TARGET_MULTI_PROFILE=n
CONFIG_TARGET_ALL_PROFILES=n
CONFIG_TARGET_DEVICE_mediatek_filogic_DEVICE_hiveton_h5000m=y

# 内核调整
CONFIG_PACKAGE_kmod-cryptodev=y
CONFIG_PACKAGE_kmod-tls=y

# H5000M 专用硬件
CONFIG_PACKAGE_luci-app-h5000m-fancontrol=y

# ===== luci-app-netmonitor：网络质量监控（自用）=====
#
# 包来源：https://github.com/LianXia233/luci-app-netmonitor（分支 main，版本 1.2.1）
# 构成：procd 常驻守护（netmon-daemon.sh）+ ucode RPC（luci.netmonitor）+
#       LuCI JS 前端，LUCI_PKGARCH=all（纯文本架构，无 Rust 编译）。
# 其 Makefile 走 feeds/luci/luci.mk 编译框架，源码由
# Scripts/Packages-NetMonitor.sh 在 make defconfig 之前就位。
#
# Makefile 的 LUCI_DEPENDS 为 +luci-base +luci-mod-status +ucode-mod-fs
# +ucode-mod-uci +ucode-mod-ubus +ucode-mod-uloop，其中 luci-base 已在
# Config/GENERAL.txt 选中；其余逐项显式选中，避免依赖链变动时静默缺包
#（做法与 NetWiz 配置一致：依赖只靠 DEPENDS 隐式解析时，一旦上游调整
# 打包或某依赖在别的配置层被置 =n，make defconfig 后会缺功能组件）。
CONFIG_PACKAGE_luci-app-netmonitor=y
CONFIG_PACKAGE_luci-i18n-netmonitor-zh-cn=y
CONFIG_PACKAGE_luci-mod-status=y
CONFIG_PACKAGE_ucode-mod-fs=y
CONFIG_PACKAGE_ucode-mod-uci=y
CONFIG_PACKAGE_ucode-mod-ubus=y
CONFIG_PACKAGE_ucode-mod-uloop=y

# ===== luci-app-taygedo：塔吉多自动签到（自用）=====
#
# 包来源：https://github.com/LianXia233/taygedo-CI（分支 main，版本 0.5.0）
# 构成：LuCI JS 前端（服务 → 塔吉多签到）+ Rust 后端 /usr/bin/taygedo-rs。
# 后端为上游预编译 musl 静态二进制：包 Makefile 的 Build/Prepare 阶段从
# GitHub Release 下载（aarch64_cortex-a53 → taygedo-rs-aarch64-unknown-linux-musl.tar.gz），
# 不经 OpenWrt rust/host 源码构建，构建耗时可控。
# DEPENDS: +ca-bundle +luci-base（luci-base 已在 GENERAL.txt 选中；
# ca-bundle 在 GENERAL.txt 中被注释，故此处显式选中）。
CONFIG_PACKAGE_luci-app-taygedo=y
CONFIG_PACKAGE_ca-bundle=y

# ===== Podman 容器管理（自用）=====
#
# 前端：https://github.com/Zerogiven-OpenWRT-Packages/luci-app-podman（v2.3.4，LuCI JS）
# 后端：packages feed 的 utils/podman（5.2.2，Go）
#
# 关键点：前端 Makefile 把 podman 只写在 PKG_BUILD_DEPENDS（构建期依赖，
# 不产出运行时依赖），LUCI_DEPENDS 里没有 +podman。因此后端必须在此显式
# 选中，否则只会装一个点不开的空界面。
#
# 硬件前提：H5000M 是 mt7987a + eMMC（mmc0 8-bit 非可插拔），空间足以放镜像；
# 容器镜像建议放 eMMC 的数据分区（/overlay 之外），不要塞进 tmpfs。
CONFIG_PACKAGE_luci-app-podman=y
CONFIG_PACKAGE_luci-i18n-podman-zh-cn=y
CONFIG_PACKAGE_podman=y

# podman 运行时依赖：feed 的 DEPENDS 已能自动解析，这里逐项显式选中，
# 与 NetWiz / netmonitor 的做法一致 —— 依赖只靠隐式解析时，一旦上游调整
# 打包或某依赖在别的配置层被置 =n，make defconfig 后会静默缺件，
# 表现为 podman 能启动但 pod / network / DNS 功能逐个报错。
CONFIG_PACKAGE_conmon=y
CONFIG_PACKAGE_crun=y
CONFIG_PACKAGE_netavark=y
CONFIG_PACKAGE_aardvark-dns=y
CONFIG_PACKAGE_catatonit=y
CONFIG_PACKAGE_slirp4netns=y
CONFIG_PACKAGE_libseccomp=y
CONFIG_PACKAGE_libgpgme=y
CONFIG_PACKAGE_nsenter=y
CONFIG_PACKAGE_zoneinfo-simple=y
# 容器网络：veth 是 feed 里 podman 的硬依赖；br-netfilter 让容器经 bridge
# 出网时能命中 NAT 规则（netavark 写的 nft 规则链）。
CONFIG_PACKAGE_kmod-veth=y
CONFIG_PACKAGE_kmod-br-netfilter=y
# OpenWrt 根文件系统本身就是 overlayfs，podman 默认 overlay driver 不能再叠
# 一层（overlay-on-overlay 不被支持），因此装 fuse-overlayfs 作为可选后端；
# 实际用哪个 driver 由 /etc/containers/storage.conf 决定。
CONFIG_PACKAGE_fuse-overlayfs=y

# 前端 rpcd / ucode 依赖（LUCI_DEPENDS 会解析，同样显式列出）
CONFIG_PACKAGE_rpcd-mod-ucode=y
CONFIG_PACKAGE_ucode-mod-socket=y
CONFIG_PACKAGE_ucode-mod-struct=y
CONFIG_PACKAGE_ucode-mod-html=y
CONFIG_PACKAGE_liblucihttp-ucode=y
CONFIG_PACKAGE_coreutils-timeout=y

# 容器网络首次开机初始化（netavark 默认走 iptables，而 25.12 镜像没有 iptables
# 二进制；aardvark-dns 还要与 dnsmasq 抢 10.88.0.1:53）。修复脚本由
# Scripts/Handles.sh 注入 files/etc/uci-defaults/98-podman-network，
# 不占包符号，此处仅作位置说明。

# MT 相关包一律交给 MT_MODE 决定：Scripts/ApplyMTMode.sh 在本层之后执行，
# 会用 =n 覆盖本文件可能写入的任何 MT 包，故本文件不声明任何 MT 包。
